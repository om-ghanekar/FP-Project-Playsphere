-- (Team Registration)
SELECT id, tournament_id, team_name, captain_name, captain_email, captain_contact, registered_at 
FROM team_registrations 
ORDER BY id DESC;

-- (Solo Pool Registration)
SELECT id, player_name, sport, preferred_position, city, contact_number, experience_level, created_at 
FROM solo_pool 
ORDER BY id DESC;

-- (Latest Solo Pool Registration)
SELECT * FROM solo_pool 
ORDER BY id DESC 
LIMIT 1;

-- (Tournament Registration / Tournament Details)
SELECT id, name, sport, city, venue, entry_fee, prize_pool, organizer_name, created_at 
FROM tournaments 
ORDER BY id DESC 
LIMIT 10;
